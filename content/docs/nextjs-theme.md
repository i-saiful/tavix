## Server And Client

In the Next.js App Router, components are **Server Components by default**. In this example, `app/layout.js` and `app/page.js` run on the server. They can read the theme cookie and send its value to another component.

`components/ThemeToggle.jsx` starts with `"use client"`, so it is a **Client Component**. It needs to run in the browser because it uses React state, handles a button click, and updates the document's theme immediately.

The server component is not called in the browser. The server page awaits the theme, then passes a simple string such as `"light"` or `"dark"` to the client component:

```jsx
const theme = await initialTheme();
return <ThemeToggle initialTheme={theme} />;
```

Do not pass the function itself:

```jsx
// Incorrect: useState treats a function as a lazy initializer and calls it.
return <ThemeToggle initialTheme={initialTheme} />;
```

`saveTheme` is a Server Action. The client component calls it after a click, and Next.js runs that action on the server to save the cookie. Calling a Server Action does not turn the client component into a Server Component.

## Theme Files ( `Dynamic` - server-rendered on demand)

### `utils/theme.js`

```js
export const THEME_COOKIE = "theme";

export const THEMES = Object.freeze({
  LIGHT: "light",
  DARK: "dark",
});

export function isTheme(value) {
  return Object.values(THEMES).includes(value);
}
```

### `actions/theme-actions.js`

```js
"use server";

import { cookies } from "next/headers";
import { isTheme, THEME_COOKIE, THEMES } from "@/utils/theme";

export async function saveTheme(theme) {
  if (!isTheme(theme)) {
    return;
  }

  const cookieStore = await cookies();
  cookieStore.set(THEME_COOKIE, theme, {
    httpOnly: true,
    maxAge: 60 * 60 * 24 * 365,
    path: "/",
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
  });
}

export async function initialTheme() {
  const cookieStore = await cookies();
  const savedTheme = cookieStore.get(THEME_COOKIE)?.value;
  const theme = savedTheme === THEMES.DARK ? THEMES.DARK : THEMES.LIGHT;
  return theme;
}
```

`initialTheme` returns the saved theme or `"light"` if no valid dark theme is saved. The layout and page await it on the server.

### `app/page.js`


The page passes the resolved theme string to the client component. The component does not receive or call `initialTheme`.

### `components/ThemeToggle.jsx`

```jsx
"use client";

import { Button } from "tavix";
import { useState } from "react";
import { THEMES } from "@/utils/theme";
import { saveTheme } from "@/actions/theme-actions";


export default function ThemeToggle({ initialTheme }) {
  const [theme, setTheme] = useState(initialTheme);

  function toggleTheme() {
    const nextTheme = theme === THEMES.LIGHT ? THEMES.DARK : THEMES.LIGHT;
    setTheme(nextTheme);
    document.documentElement.dataset.theme = nextTheme;
    void saveTheme(nextTheme);
  }

  return (
    <div className="flex justify-center items-center h-screen flex-col gap-5 bg-canvas">
      <h1 className="h1 text-primary">
        Currently Theme is
        <span className="font-monospace bg-surface px-3 rounded ml-4 letter-spacing-widest">
          {theme === THEMES.LIGHT ? "Light theme" : "Dark theme"}
        </span>
      </h1>
      <Button className="btn btn-primary" onClick={toggleTheme}>
        Theme Change
      </Button>
    </div>
  );
}
```

On click, this component updates its state and the page immediately, then asks the server action to persist the choice in a cookie.

### persistence.js
```js
const THEME_KEY = "theme";

const THEMES = Object.freeze({
  DARK: "dark",
  LIGHT: "light",
});

export function toggleTheme() {
  const currentTheme = document.documentElement.dataset.theme;
  const nextTheme = currentTheme === THEMES.DARK ? THEMES.LIGHT : THEMES.DARK;
  document.documentElement.dataset.theme = nextTheme;
  window.localStorage.setItem(THEME_KEY, nextTheme);
}
```

### theme-init.js
```js
try {
  document.documentElement.dataset.theme = localStorage.getItem("theme") === "dark" ? "dark" : "light";
} catch {
  document.documentElement.dataset.theme = "light";
}
```

## Another Approce

### persistence.js

```js
const THEME_KEY = "theme";

function getLocalStorage(key) {
  try {
    return window.localStorage.getItem(key);
  } catch {
    return null;
  }
}

function setLocalStorage(key, value) {
  try {
    window.localStorage.setItem(key, value);
  } catch {}
}

export const THEMES = Object.freeze({
  DARK: "dark",
  LIGHT: "light",
});

export function initTheme() {
  const theme = getLocalStorage(THEME_KEY) === THEMES.LIGHT ? THEMES.LIGHT : THEMES.DARK;
  document.documentElement.dataset.theme = theme;
  return theme;
}

export function saveTheme(theme) {
  document.documentElement.dataset.theme = theme;
  setLocalStorage(THEME_KEY, theme);
}
```
