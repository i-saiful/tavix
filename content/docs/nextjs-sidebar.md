# Next.js Client Component Sidebar Using Localhost

Use the `Sidebar` component to build a navigation rail or side menu with grouped links, nested sections, and a footer area.

## Source Structure

```folder
project/
|-- app/
    +-- layout.jsx
|-- components/
    +-- Sidebar.jsx
|-- utils/
    +-- persistence.js
```

```jsx

const items = {

        { id: "media", title: "Media", icon: "image", type: "link", url: "/content/media" },
        { id: "categories", title: "Categories", icon: "folder", type: "link", url: "/content/categories" },
      ],
    },
    {
      id: "users",
      title: "Users & Access",
      icon: "user",
      type: "link",
      url: "/users",
      children: [
        { id: "all-users", title: "All Users", icon: "users", type: "link", url: "/users/all" },
        { id: "roles", title: "Roles", icon: "shield", type: "link", url: "/users/roles" },
      ],
    },
    { id: "analytics", title: "Analytics", icon: "file", type: "link", url: "/analytics" },
    { id: "settings", title: "Settings", icon: "settings", type: "link", url: "/settings" },
  ],
  footer: [
    { id: "help", title: "Help Center", icon: "question", type: "link", url: "/help" },
    { id: "sign-out", title: "Sign Out", icon: "log-out", type: "action", action: "/logout" },
  ],
};

export default function AppSidebar() {
  const [selection, setSelection] = useState({ activeId: null, expandedId: null });

  useMount(() => {
    setSelection(initSidebar());
  });

  function handleSelectionChange(nextSelection) {
    setSelection(nextSelection);
    saveSidebar(nextSelection);
  }

  return (
    <Sidebar
      navItems={items}
      Link={Link}
      activeId={selection.activeId}
      expandedId={selection.expandedId}
      onSelectionChange={handleSelectionChange}
      onAction={(action) => {
        if (action === "/logout") {
          window.location.assign(action);
        }
      }}
      enableTheme={true}
      curved={false}
    />
  );
}
```

## `persistence.js`

```jsx
const SIDEBAR_KEY = "sidebar";

export function initSidebar() {
  const savedSelection = getLocalStorage(SIDEBAR_KEY);
  if (!savedSelection) {
    return { activeId: null, expandedId: null };
  }

  try {
    const selection = JSON.parse(savedSelection);
    return {
      activeId: typeof selection?.activeId === "string" ? selection.activeId : null,
      expandedId: typeof selection?.expandedId === "string" ? selection.expandedId : null,
    };
  } catch {
    return { activeId: null, expandedId: null };
  }
}

export function saveSidebar(selection) {
  return setLocalStorage(SIDEBAR_KEY, JSON.stringify(selection));
}
```

## `sidebar-actions.jsx`

```js
"use server";

import { cookies } from "next/headers";
import { isSelection, parseSelection, SIDEBAR_COOKIE } from "@/utils/sidebar";

export async function getSelection() {
  const cookieStore = await cookies();
  return parseSelection(cookieStore.get(SIDEBAR_COOKIE)?.value);
}

export async function saveSelection(selection) {
  if (!isSelection(selection)) {
    return;
  }

  const cookieStore = await cookies();
  cookieStore.set(SIDEBAR_COOKIE, JSON.stringify(selection), {
    httpOnly: true,
    maxAge: 60 * 60 * 24 * 365,
    path: "/",
    sameSite: "lax",
    secure: process.env.NODE_ENV === "production",
  });
}
```

## sidebar.js

```js
export const SIDEBAR_COOKIE = "sidebar";

export const DEFAULT_SELECTION = Object.freeze({
  activeId: null,
  expandedId: null,
});

export function isSelection(value) {
  return (
    value !== null &&
    typeof value === "object" &&
    !Array.isArray(value) &&
    (value.activeId === null || typeof value.activeId === "string") &&
    (value.expandedId === null || typeof value.expandedId === "string")
  );
}

export function parseSelection(value) {
  if (!value) {
    return DEFAULT_SELECTION;
  }

  try {
    const selection = JSON.parse(value);
    return isSelection(selection) ? selection : DEFAULT_SELECTION;
  } catch {
    return DEFAULT_SELECTION;
  }
}
```
